/* ---------- log backups --------------------------------------------------- */
CREATE   PROCEDURE dbo.AFS_BackupLogAll
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @db sysname, @file nvarchar(400), @sql nvarchar(max), @msg nvarchar(400);

    DECLARE cur CURSOR LOCAL FAST_FORWARD FOR
        SELECT name FROM sys.databases
        WHERE name LIKE N'AFS[_]%' AND state_desc = N'ONLINE'
          AND recovery_model_desc = N'FULL';
    OPEN cur; FETCH NEXT FROM cur INTO @db;
    WHILE @@FETCH_STATUS = 0
    BEGIN
        SET @file = N'C:\ProgramData\AFS\backups\log\' + @db + N'_'
                  + FORMAT(SYSUTCDATETIME(), N'yyyyMMdd_HHmmss') + N'.trn';
        SET @sql = N'BACKUP LOG ' + QUOTENAME(@db) + N' TO DISK = '
                 + QUOTENAME(@file, CHAR(39)) + N' WITH INIT, COMPRESSION, CHECKSUM;';
        EXEC (@sql);
        SET @msg = N'log backup written to ' + @file;
        INSERT AFS_Audit.dbo.DbaOpsLog (Source, Severity, DatabaseName, Message)
        VALUES (N'AFS_BackupLogAll', N'INFO', @db, @msg);
        FETCH NEXT FROM cur INTO @db;
    END
    CLOSE cur; DEALLOCATE cur;
END
